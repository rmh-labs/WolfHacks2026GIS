CREATE QUERY getEvacuationRoute(VERTEX<Intersection> startNode, VERTEX<Intersection> targetNode) {
    /* TigerGraph Dynamic Pathfinding with Hazard Penalty */
    HeapAccum<PathTuple>(1, minDistance ASC) @@minPath;
    
    Start = {startNode};
    
    WHILE Start.size() > 0 DO
        Start = SELECT t FROM Start:s -(RoadEdge:e)- Intersection:t
                ACCUM 
                    // Dynamic Edge Weight calculation
                    DOUBLE edgeCost = e.distance * (1 + e.floodHazardLevel * 100),
                    t.@minDistance += s.@minDistance + edgeCost;
    END;
    PRINT @@minPath;
}
